# @file
# @brief Configure native C17 targets, tests, warnings, and platform-specific archives.
cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum supported macOS version")
endif()
project(DiskSan VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(DISKSAN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(DISKSAN_WERROR "Treat compiler warnings as errors" ON)
include(CTest)
configure_file(include/disksan/version.h.in generated/disksan/version.h @ONLY)

# Share warning and feature settings without exposing platform APIs to the core.
add_library(disksan_options INTERFACE)
if(NOT WIN32)
  target_compile_definitions(disksan_options INTERFACE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
else()
  target_compile_definitions(disksan_options INTERFACE _WIN32_WINNT=0x0601 WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
target_include_directories(disksan_options INTERFACE include "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(disksan_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(DISKSAN_WERROR)
    target_compile_options(disksan_options INTERFACE -Werror)
  endif()
  if(DISKSAN_SANITIZERS)
    target_compile_options(disksan_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(disksan_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
if(MSVC)
  target_compile_options(disksan_options INTERFACE /W4)
  if(DISKSAN_WERROR)
    target_compile_options(disksan_options INTERFACE /WX)
  endif()
endif()

add_library(disksan_core STATIC src/core/device.c src/core/classification.c src/core/assessment.c
  src/core/target_compare.c src/core/capabilities.c src/core/usage.c
  src/output/output.c src/output/assessment_output.c)
target_link_libraries(disksan_core PUBLIC disksan_options)

# Select exactly one native adapter for the target operating system.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
  add_library(disksan_platform STATIC src/platform/linux/discovery.c
    src/platform/linux/block_info.c src/platform/linux/device_info.c
    src/platform/linux/usage.c src/platform/linux/usage_parse.c)
  target_link_libraries(disksan_platform PUBLIC disksan_core PkgConfig::UDEV)
elseif(APPLE)
  target_compile_definitions(disksan_options INTERFACE _DARWIN_C_SOURCE)
  add_library(disksan_platform STATIC src/platform/macos/discovery.c src/platform/macos/usage.c)
  target_link_libraries(disksan_platform PUBLIC disksan_core "-framework IOKit" "-framework CoreFoundation" "-framework DiskArbitration")
elseif(WIN32)
  add_library(disksan_platform STATIC src/platform/windows/discovery.c src/platform/windows/metadata.c src/platform/windows/usage.c)
  target_link_libraries(disksan_platform PUBLIC disksan_core setupapi)
else()
  message(STATUS "No native backend for ${CMAKE_SYSTEM_NAME}: building portable tests only")
  add_library(disksan_platform STATIC src/platform/unsupported.c)
  target_link_libraries(disksan_platform PUBLIC disksan_core)
endif()
# The native window depends on discovery/usage; keep it with the adapter to avoid a static-library cycle.
target_sources(disksan_platform PRIVATE src/core/target.c)
add_executable(disksan src/main.c)
target_link_libraries(disksan PRIVATE disksan_platform)
# Ship exactly the tested executable; never embed private dependency directories.
set_target_properties(disksan PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "")
if(MINGW)
  target_link_options(disksan PRIVATE -municode -static-libgcc)
endif()

if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  # Enforce the source documentation standard alongside behavior tests.
  add_test(NAME documentation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_documentation.py")
  add_executable(test_core tests/test_core.c)
  target_link_libraries(test_core PRIVATE disksan_core)
  add_executable(fixture_output tests/fixture_output.c)
  target_link_libraries(fixture_output PRIVATE disksan_core)
  add_test(NAME core COMMAND test_core)
  add_executable(test_assessment tests/test_assessment.c)
  target_link_libraries(test_assessment PRIVATE disksan_core)
  add_test(NAME assessment_policy COMMAND test_assessment)
  set_tests_properties(assessment_policy PROPERTIES LABELS "portable;assessment")
  add_executable(test_target tests/test_target.c src/core/target.c)
  target_link_libraries(test_target PRIVATE disksan_core)
  add_test(NAME target_window COMMAND test_target)
  set_tests_properties(target_window PROPERTIES LABELS "portable;assessment;fault")
  add_executable(test_usage_parse tests/test_usage_parse.c src/platform/linux/usage_parse.c)
  target_link_libraries(test_usage_parse PRIVATE disksan_core)
  add_test(NAME usage_parsers COMMAND test_usage_parse)
  set_tests_properties(usage_parsers PROPERTIES LABELS "portable;assessment")
  add_executable(test_usage_graph tests/test_usage_graph.c)
  target_link_libraries(test_usage_graph PRIVATE disksan_core)
  add_test(NAME usage_graph COMMAND test_usage_graph)
  set_tests_properties(usage_graph PROPERTIES LABELS "portable;assessment")
  # Instrument only a separate core object, keeping production allocation unchanged.
  add_library(disksan_fault_core OBJECT src/core/device.c)
  target_link_libraries(disksan_fault_core PRIVATE disksan_options)
  if(MSVC)
    target_compile_options(disksan_fault_core PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/tests/allocator_override.h")
  else()
    target_compile_options(disksan_fault_core PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/tests/allocator_override.h")
  endif()
  add_executable(test_identity tests/test_identity.c $<TARGET_OBJECTS:disksan_fault_core>)
  target_link_libraries(test_identity PRIVATE disksan_options)
  add_test(NAME identity_faults COMMAND test_identity)
  add_executable(test_windows_metadata tests/test_windows_metadata.c src/platform/windows/metadata.c)
  target_link_libraries(test_windows_metadata PRIVATE disksan_options)
  add_test(NAME windows_metadata COMMAND test_windows_metadata)
  set_tests_properties(windows_metadata PROPERTIES LABELS "portable;fault")
  set_tests_properties(identity_faults PROPERTIES LABELS "fault;portable")
  add_test(NAME output_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_output.py" $<TARGET_FILE:fixture_output>)
  add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" $<TARGET_FILE:disksan> "${CMAKE_SYSTEM_NAME}")
  add_executable(fixture_assessment_cli src/main.c src/core/target.c tests/fixture_assessment_cli.c)
  target_link_libraries(fixture_assessment_cli PRIVATE disksan_core)
  if(MINGW)
    target_link_options(fixture_assessment_cli PRIVATE -municode -static-libgcc)
  endif()
  add_executable(fixture_assessment_output tests/fixture_assessment_output.c)
  target_link_libraries(fixture_assessment_output PRIVATE disksan_core)
  add_test(NAME assessment_cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_assess_cli.py"
    $<TARGET_FILE:disksan> $<TARGET_FILE:fixture_assessment_cli> $<TARGET_FILE:fixture_assessment_output> "${CMAKE_SYSTEM_NAME}")
  set_tests_properties(assessment_cli PROPERTIES LABELS "portable;assessment")
  set_tests_properties(documentation core output_contract cli PROPERTIES LABELS "portable")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR APPLE OR WIN32)
    add_test(NAME assessment_native COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_assessment_native.py"
      $<TARGET_FILE:disksan> "${CMAKE_SYSTEM_NAME}")
    set_tests_properties(assessment_native PROPERTIES LABELS "integration;read-only;assessment")
  endif()
  if(APPLE)
    add_library(macos_usage_fault OBJECT src/platform/macos/usage.c)
    target_link_libraries(macos_usage_fault PRIVATE disksan_options)
    target_compile_options(macos_usage_fault PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/tests/macos_usage_override.h")
    add_executable(test_macos_usage tests/test_macos_usage.c $<TARGET_OBJECTS:macos_usage_fault>)
    target_link_libraries(test_macos_usage PRIVATE disksan_core "-framework DiskArbitration" "-framework CoreFoundation")
    add_test(NAME macos_usage_faults COMMAND test_macos_usage)
    set_tests_properties(macos_usage_faults PROPERTIES LABELS "macos;assessment;fault")
    add_test(NAME macos_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_macos.py" $<TARGET_FILE:disksan>)
    set_tests_properties(macos_integration PROPERTIES LABELS "integration;read-only")
  endif()
  if(WIN32)
    add_library(windows_usage_fault OBJECT src/platform/windows/usage.c)
    target_link_libraries(windows_usage_fault PRIVATE disksan_options)
    if(MSVC)
      target_compile_options(windows_usage_fault PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/tests/windows_usage_override.h")
    else()
      target_compile_options(windows_usage_fault PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/tests/windows_usage_override.h")
    endif()
    add_executable(test_windows_usage tests/test_windows_usage.c src/platform/windows/metadata.c $<TARGET_OBJECTS:windows_usage_fault>)
    target_link_libraries(test_windows_usage PRIVATE disksan_core)
    add_test(NAME windows_usage_faults COMMAND test_windows_usage)
    set_tests_properties(windows_usage_faults PROPERTIES LABELS "portable;windows;assessment;fault")
    add_test(NAME windows_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_windows.py" $<TARGET_FILE:disksan>)
    set_tests_properties(windows_integration PROPERTIES LABELS "integration;read-only")
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(test_linux_usage tests/test_linux_usage.c src/platform/linux/usage.c src/platform/linux/usage_parse.c)
    target_link_libraries(test_linux_usage PRIVATE disksan_core)
    target_link_options(test_linux_usage PRIVATE -Wl,--wrap=open -Wl,--wrap=fstat -Wl,--wrap=fstatfs
      -Wl,--wrap=stat -Wl,--wrap=fdopen -Wl,--wrap=close -Wl,--wrap=fclose
      -Wl,--wrap=ds_strings_add -Wl,--wrap=fgetc -Wl,--wrap=ferror)
    add_test(NAME linux_usage_faults COMMAND test_linux_usage)
    set_tests_properties(linux_usage_faults PROPERTIES LABELS "linux;assessment;fault")
    add_executable(test_block_info tests/test_block_info.c src/platform/linux/block_info.c)
    target_link_libraries(test_block_info PRIVATE disksan_core)
    target_link_options(test_block_info PRIVATE -Wl,--wrap=open -Wl,--wrap=fstat -Wl,--wrap=ioctl -Wl,--wrap=close)
    add_test(NAME block_info_errors COMMAND test_block_info)
    add_executable(test_discovery_errors tests/test_discovery_errors.c)
    target_link_libraries(test_discovery_errors PRIVATE disksan_platform)
    target_link_options(test_discovery_errors PRIVATE
      -Wl,--wrap=udev_new -Wl,--wrap=udev_enumerate_new
      -Wl,--wrap=udev_enumerate_add_match_subsystem -Wl,--wrap=udev_enumerate_scan_devices
      -Wl,--wrap=udev_enumerate_get_list_entry -Wl,--wrap=udev_list_entry_get_name
      -Wl,--wrap=udev_list_entry_get_next -Wl,--wrap=udev_device_new_from_syspath -Wl,--wrap=ds_linux_collect -Wl,--wrap=udev_device_unref)
    add_test(NAME discovery_errors COMMAND test_discovery_errors)
    add_executable(test_linux_metadata tests/test_linux_metadata.c)
    target_link_libraries(test_linux_metadata PRIVATE disksan_platform)
    target_link_options(test_linux_metadata PRIVATE
      -Wl,--wrap=udev_device_get_syspath -Wl,--wrap=udev_device_get_subsystem
      -Wl,--wrap=udev_device_get_devnum -Wl,--wrap=udev_device_get_devnode
      -Wl,--wrap=udev_device_get_devtype -Wl,--wrap=udev_device_get_property_value
      -Wl,--wrap=udev_device_get_parent -Wl,--wrap=udev_device_get_devlinks_list_entry
      -Wl,--wrap=ds_linux_probe)
    add_test(NAME linux_metadata_faults COMMAND test_linux_metadata)
    set_tests_properties(linux_metadata_faults PROPERTIES LABELS "fault;linux")
    set_tests_properties(block_info_errors discovery_errors PROPERTIES LABELS "fault;linux")
    add_test(NAME linux_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_linux.py" $<TARGET_FILE:disksan>)
    set_tests_properties(linux_integration PROPERTIES LABELS "integration;read-only")
  endif()
endif()
# Package only installed runtime files and public documentation.
set(CPACK_PACKAGE_NAME "disksan")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
if(APPLE)
  set(DISKSAN_PACKAGE_OS macos)
elseif(WIN32)
  set(DISKSAN_PACKAGE_OS windows)
else()
  string(TOLOWER "${CMAKE_SYSTEM_NAME}" DISKSAN_PACKAGE_OS)
endif()
set(DISKSAN_PACKAGE_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
if(WIN32 AND CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
  set(DISKSAN_PACKAGE_ARCH x86_64)
endif()
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
  string(REPLACE ";" "-" DISKSAN_PACKAGE_ARCH "${CMAKE_OSX_ARCHITECTURES}")
endif()
set(CPACK_PACKAGE_FILE_NAME "disksan-${PROJECT_VERSION}-${DISKSAN_PACKAGE_OS}-${DISKSAN_PACKAGE_ARCH}")
if(WIN32 OR APPLE)
  set(CPACK_GENERATOR ZIP)
else()
  set(CPACK_GENERATOR TGZ)
endif()
install(FILES README.md CHANGELOG.md CONTRIBUTING.md SECURITY.md LICENSE DESTINATION share/doc/disksan)
install(DIRECTORY docs/ DESTINATION share/doc/disksan/docs FILES_MATCHING PATTERN "*.md" PATTERN "FILE_MANIFEST.txt")
include(CPack)
if(BUILD_TESTING)
  add_test(NAME packaging_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_package.py"
    --build "${CMAKE_CURRENT_BINARY_DIR}" --binary $<TARGET_FILE:disksan> --version "${PROJECT_VERSION}" --config $<CONFIG>)
  set_tests_properties(packaging_smoke PROPERTIES LABELS "packaging")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR APPLE OR WIN32)
  install(TARGETS disksan RUNTIME DESTINATION bin)
endif()
