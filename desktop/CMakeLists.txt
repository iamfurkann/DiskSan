# @file
# @brief Build optional Qt desktop components separately from the standalone C CLI and its archives.

# Qt 6.12 desktop needs macOS 14.4; the parent C CLI retains its independent target.
if(APPLE AND CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "14.4")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.4")
endif()
enable_language(CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Quick Qml QuickControls2)

add_library(disksan_desktop_contracts STATIC backend/BoundedJson.cpp backend/CliContract.cpp)
target_link_libraries(disksan_desktop_contracts PUBLIC Qt6::Core)
target_include_directories(disksan_desktop_contracts PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/backend")
target_include_directories(disksan_desktop_contracts PRIVATE "${PROJECT_BINARY_DIR}/generated")
add_library(disksan_desktop_service STATIC backend/DeviceService.cpp backend/DeviceService.h)
set_target_properties(disksan_desktop_service PROPERTIES AUTOMOC ON)
target_link_libraries(disksan_desktop_service PUBLIC disksan_desktop_contracts)
if(MSVC)
  target_compile_options(disksan_desktop_contracts PUBLIC /W4)
  if(DISKSAN_WERROR)
    target_compile_options(disksan_desktop_contracts PUBLIC /WX)
  endif()
else()
  target_compile_options(disksan_desktop_contracts PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(DISKSAN_WERROR)
    target_compile_options(disksan_desktop_contracts PUBLIC -Werror)
  endif()
  if(DISKSAN_SANITIZERS)
    target_compile_options(disksan_desktop_contracts PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(disksan_desktop_contracts PUBLIC -fsanitize=address,undefined)
  endif()
endif()
qt_add_executable(disksan-desktop MACOSX_BUNDLE main.cpp)
target_link_libraries(disksan-desktop PRIVATE disksan_desktop_service Qt6::Quick Qt6::Qml Qt6::QuickControls2)
target_include_directories(disksan-desktop PRIVATE "${PROJECT_BINARY_DIR}/generated")
target_compile_definitions(disksan-desktop PRIVATE "DISKSAN_CLI_PATH=\"$<TARGET_FILE:disksan>\"")
add_dependencies(disksan-desktop disksan)
qt_add_resources(disksan-desktop desktop_ui PREFIX "/disksan"
  FILES qml/Main.qml qml/ReadOnlyLabel.qml qml/EvidenceList.qml)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test_desktop_json tests/test_bounded_json.cpp)
  target_link_libraries(test_desktop_json PRIVATE disksan_desktop_contracts)
  add_test(NAME desktop_json_bounds COMMAND test_desktop_json)
  set_tests_properties(desktop_json_bounds PROPERTIES LABELS "desktop;fault")
  add_executable(test_desktop_contract tests/test_cli_contract.cpp)
  target_link_libraries(test_desktop_contract PRIVATE disksan_desktop_contracts)
  add_test(NAME desktop_cli_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli_contract.py"
    $<TARGET_FILE:test_desktop_contract> $<TARGET_FILE:fixture_plan_cli> $<TARGET_FILE:disksan-desktop>)
  set_tests_properties(desktop_cli_contract PROPERTIES LABELS "desktop;fault")
  set(process_fixture_dir "${CMAKE_CURRENT_BINARY_DIR}/process-fixtures")
  add_custom_command(OUTPUT "${process_fixture_dir}/list.json"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/generate_process_fixtures.py"
      $<TARGET_FILE:fixture_plan_cli> "${process_fixture_dir}"
    DEPENDS fixture_plan_cli tests/generate_process_fixtures.py VERBATIM)
  add_custom_target(desktop_process_data DEPENDS "${process_fixture_dir}/list.json")
  add_executable(fixture_desktop_process tests/fixture_process.cpp)
  target_link_libraries(fixture_desktop_process PRIVATE Qt6::Core)
  target_compile_definitions(fixture_desktop_process PRIVATE "DISKSAN_FIXTURE_DIR=\"${process_fixture_dir}\"")
  add_dependencies(fixture_desktop_process desktop_process_data)
  add_executable(test_desktop_service tests/test_device_service.cpp)
  set_target_properties(test_desktop_service PROPERTIES AUTOMOC ON)
  target_link_libraries(test_desktop_service PRIVATE disksan_desktop_service Qt6::Test)
  target_compile_definitions(test_desktop_service PRIVATE "DISKSAN_PROCESS_FIXTURE=\"$<TARGET_FILE:fixture_desktop_process>\"")
  add_dependencies(test_desktop_service fixture_desktop_process)
  add_test(NAME desktop_process_boundary COMMAND test_desktop_service)
  set_tests_properties(desktop_process_boundary PROPERTIES LABELS "desktop;fault" TIMEOUT 60)
  qt_add_executable(disksan-desktop-fixture MACOSX_BUNDLE main.cpp)
  target_link_libraries(disksan-desktop-fixture PRIVATE disksan_desktop_service Qt6::Quick Qt6::Qml Qt6::QuickControls2)
  target_include_directories(disksan-desktop-fixture PRIVATE "${PROJECT_BINARY_DIR}/generated")
  target_compile_definitions(disksan-desktop-fixture PRIVATE DISKSAN_DESKTOP_FIXTURE
    "DISKSAN_CLI_PATH=\"$<TARGET_FILE:fixture_desktop_process>\"")
  add_dependencies(disksan-desktop-fixture fixture_desktop_process)
  qt_add_resources(disksan-desktop-fixture desktop_fixture_ui PREFIX "/disksan"
    FILES qml/Main.qml qml/ReadOnlyLabel.qml qml/EvidenceList.qml)
  add_executable(test_desktop_ui tests/test_desktop_ui.cpp)
  set_target_properties(test_desktop_ui PROPERTIES AUTOMOC ON)
  target_link_libraries(test_desktop_ui PRIVATE disksan_desktop_service Qt6::Quick Qt6::Qml Qt6::QuickControls2 Qt6::Test)
  target_include_directories(test_desktop_ui PRIVATE "${PROJECT_BINARY_DIR}/generated")
  target_compile_definitions(test_desktop_ui PRIVATE
    "DISKSAN_PROCESS_FIXTURE=\"$<TARGET_FILE:fixture_desktop_process>\"" "DISKSAN_NATIVE_CLI=\"$<TARGET_FILE:disksan>\"")
  add_dependencies(test_desktop_ui fixture_desktop_process disksan)
  qt_add_resources(test_desktop_ui desktop_test_ui PREFIX "/disksan"
    FILES qml/Main.qml qml/ReadOnlyLabel.qml qml/EvidenceList.qml)
  add_test(NAME desktop_ui_fixture COMMAND test_desktop_ui fixtureJourney invalidStates keyboardAndAccessibility selectedFailure)
  add_test(NAME desktop_ui_native COMMAND test_desktop_ui nativeInventory)
  # Four independent UI journeys share this CTest process, including instrumented child startups.
  set_tests_properties(desktop_ui_fixture PROPERTIES LABELS "desktop;gui" TIMEOUT 180
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;DISKSAN_TEST_SCREENSHOT=${CMAKE_CURRENT_BINARY_DIR}/desktop-demo.png")
  set_tests_properties(desktop_ui_native PROPERTIES LABELS "desktop;gui;integration;read-only" TIMEOUT 90
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
